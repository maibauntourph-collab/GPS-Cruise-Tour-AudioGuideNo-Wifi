  <div class="info-box">
    📍 <b>Rome GPS Audio Guide</b><br>
    반경 50m 이내 접근 시 자동 음성 가이드 재생<br>
    <button id="clearRoute" class="route-btn">🧭 길찾기 초기화</button>
  </div>

  <script src="https://unpkg.com/leaflet/dist/leaflet.js"></script>
  <script src="https://unpkg.com/@geoman-io/leaflet-geoman-free@2.13.0/dist/leaflet-geoman.min.js"></script>
  <script src="https://unpkg.com/leaflet-routing-machine/dist/leaflet-routing-machine.js"></script>

  <script>
    // ===============================
    //  명소 데이터
    // ===============================
    const pois = [
      { id: 'colosseum', name: '콜로세움', lat: 41.8902, lng: 12.4922, radius: 70,
        narration: '로마의 상징, 콜로세움입니다. 검투사들의 격투가 열리던 고대 원형 경기장이죠.' },
      { id: 'roman_forum', name: '로마 포룸', lat: 41.8925, lng: 12.4853, radius: 60,
        narration: '로마의 정치 중심지, 포룸입니다. 공화국 시대의 심장부로, 역사적 건물들이 남아있습니다.' },
      { id: 'trevi_fountain', name: '트레비 분수', lat: 41.9009, lng: 12.4833, radius: 50,
        narration: '트레비 분수입니다. 동전을 던지면 로마로 다시 돌아온다는 전설이 있습니다.' },
      { id: 'pantheon', name: '판테온', lat: 41.8986, lng: 12.4768, radius: 50,
        narration: '판테온은 모든 신들에게 바쳐진 신전으로, 고대 건축의 걸작입니다.' },
      { id: 'spanish_steps', name: '스페인 광장', lat: 41.9059, lng: 12.4823, radius: 50,
        narration: '스페인 광장은 영화 "로마의 휴일" 촬영지로 유명합니다.' }
    ];

    // ===============================
    //  지도 (OpenStreetMap + Leaflet)
    // ===============================
    const map = L.map('map').setView([41.8902, 12.4922], 14);
    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
      attribution: '&copy; OpenStreetMap contributors'
    }).addTo(map);

    // 명소 마커 추가
    pois.forEach(p => {
      const marker = L.marker([p.lat, p.lng]).addTo(map);
      marker.bindPopup(`<b>${p.name}</b><br><button onclick="setRoute(${p.lat},${p.lng})">이곳으로 길찾기</button>`);
    });

    // ===============================
    //  오디오 가이드 (Web Speech API)
    // ===============================
    const spoken = new Set();
    function speak(text) {
      const msg = new SpeechSynthesisUtterance(text);
      msg.lang = 'ko-KR';
      msg.rate = 1.0;
      window.speechSynthesis.speak(msg);
    }

    // ===============================
    //  거리 계산 (Haversine)
    // ===============================
    function getDistance(lat1, lon1, lat2, lon2) {
      const R = 6371000;
      const toRad = v => v * Math.PI / 180;
      const dLat = toRad(lat2 - lat1);
      const dLon = toRad(lon2 - lon1);
      const a = Math.sin(dLat/2)**2 + Math.cos(toRad(lat1))*Math.cos(toRad(lat2))*Math.sin(dLon/2)**2;
      const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
      return R * c;
    }

    // ===============================
    //  GPS 추적
    // ===============================
    if (navigator.geolocation) {
      navigator.geolocation.watchPosition(pos => {
        const { latitude, longitude } = pos.coords;
        pois.forEach(p => {
          const dist = getDistance(latitude, longitude, p.lat, p.lng);
          if (dist < p.radius && !spoken.has(p.id)) {
            spoken.add(p.id);
            speak(p.narration);
          }
        });
      }, err => console.error(err), { enableHighAccuracy: true });
    }

    // ===============================
    //  길찾기 기능 (무료 API)
    // ===============================
    let routeControl = null;
    function setRoute(destLat, destLng) {
      if (routeControl) map.removeControl(routeControl);
      routeControl = L.Routing.control({
        waypoints: [
          L.latLng(map.getCenter().lat, map.getCenter().lng),
          L.latLng(destLat, destLng)
        ],
        router: L.Routing.openrouteservice('5b3ce3597851110001cf62483f9e12cf7b6f4f9eaeef70f0d9e18f17'),
        show: false,
        addWaypoints: false
      }).addTo(map);
    }

    // 초기화 버튼
    document.getElementById("clearRoute").onclick = () => {
      if (routeControl) map.removeControl(routeControl);
      routeControl = null;
    };
  </script>

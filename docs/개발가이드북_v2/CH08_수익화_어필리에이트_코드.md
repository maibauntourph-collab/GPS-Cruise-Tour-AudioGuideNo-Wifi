```typescript
// GetYourGuide — 글로벌 액티비티
export const getGYGUrl = (searchQuery: string, lang: string = 'en') => {
  const url = new URL('https://www.getyourguide.com/search/');
  url.searchParams.set('q', searchQuery);
  const partnerId = AFFILIATE_IDS.getYourGuide;
  if (partnerId !== 'YOUR_GYG_PARTNER_ID') {
    url.searchParams.set('partner_id', partnerId);
  }
  return url.toString();
};

// Viator — TripAdvisor 계열
export const getViatorUrl = (searchQuery: string) => {
  const url = new URL(`https://www.viator.com/search/${encodeURIComponent(searchQuery)}`);
  url.searchParams.set('text', searchQuery);
  return url.toString();
};

// Klook — 아시아 FIT 최강
export const getKlookUrl = (searchQuery: string, lang: string = 'en-US') => {
  const klookLang = getKlookLang(lang);
  const url = new URL(`https://www.klook.com/${klookLang}/search/`);
  url.searchParams.set('keyword', searchQuery);
  return url.toString();
};

// Trip.com — 중국 관광객 타깃
export const getTripUrl = (searchQuery: string) => {
  const url = new URL('https://www.trip.com/search/');
  url.searchParams.set('query', searchQuery);
  return url.toString();
};
```

## 8.3 다국어 라우팅

각 플랫폼은 언어별 URL 구조가 다릅니다:

```typescript
// Klook — 경로에 언어 코드 포함
export const getKlookLang = (selectedLanguage: string): string => {
  const mapping: Record<string, string> = {
    'ko': 'ko', 'ja': 'ja', 'zh': 'zh-CN',
    'th': 'th-TH', 'vi': 'vi-VN', 'id': 'id-ID'
  };
  return mapping[selectedLanguage] || 'en-US';
};
// 결과: https://www.klook.com/ko/search/?keyword=...

// Viator — locale 파라미터
export const getViatorLang = (selectedLanguage: string): string => {
  const mapping: Record<string, string> = {
    'es': 'es-ES', 'fr': 'fr-FR', 'de': 'de-DE',
    'it': 'it-IT', 'pt': 'pt-BR'
  };
  return mapping[selectedLanguage] || 'en-US';
};

// GetYourGuide — 서브도메인 스타일
export const getGYGLang = (selectedLanguage: string): string => {
  const mapping: Record<string, string> = {
    'es': 'es', 'fr': 'fr', 'de': 'de',
    'it': 'it', 'pt': 'pt-BR'
  };
  return mapping[selectedLanguage] || 'en';
```

## 8.4 자동 어필리에이트 파라미터 삽입

기존 URL의 도메인을 감지하여 **적절한 파트너 ID를 자동 삽입**:

```typescript
export const injectAffiliateParam = (
  url: string,
  settings?: AffiliateSettings
): string => {
  if (!url || url === '#') return url;

  try {
    const u = new URL(url);
    const h = u.hostname.toLowerCase();

    // 도메인 감지 → 파라미터 자동 삽입
    if (h.includes('viator.com')) {
      u.searchParams.set('pid', settings?.viatorId || AFFILIATE_IDS.viator);
    } else if (h.includes('klook.com')) {
      u.searchParams.set('aid', settings?.klookId || AFFILIATE_IDS.klook);
    } else if (h.includes('getyourguide.com')) {
      u.searchParams.set('partner_id', settings?.gygId || AFFILIATE_IDS.getYourGuide);
    } else if (h.includes('trip.com')) {
      u.searchParams.set('allianceid', settings?.tripId || AFFILIATE_IDS.trip);
    } else if (h.includes('myrealtrip.com')) {
      u.searchParams.set('partner_id', settings?.myrealtripId || AFFILIATE_IDS.myrealtrip);
    }

    return u.toString();
  } catch {
    return url; // URL 파싱 실패 시 원본 반환
  }
};
```

**사용 예시:**
```typescript
// 랜드마크에 이미 예약 URL이 저장되어 있을 때
const originalUrl = "https://www.klook.com/activity/12345-colosseum-tour";
const affiliateUrl = injectAffiliateParam(originalUrl);
// → "https://www.klook.com/activity/12345-colosseum-tour?aid=YOUR_KLOOK_AID"

// window.open으로 새 탭 열기 (앱 이탈 최소화)
window.open(affiliateUrl, '_blank', 'noopener');
```
